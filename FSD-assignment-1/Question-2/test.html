const express = require("express"); const mongoose = require("mongoose"); const multer = require("multer"); const path = require("path"); const fs = require("fs"); const app = express(); app.use(express.urlencoded({ extended: true })); app.use(express.json()); // Make uploaded photos visible in browser app.use("/uploads", express.static("uploads")); // Create uploads folder if it doesn't exist if (!fs.existsSync("uploads")) { fs.mkdirSync("uploads"); } // ================= MONGODB ================= mongoose.connect("mongodb://127.0.0.1:27017/studentDB") .then(() => console.log("MongoDB Connected")) .catch(err => console.log(err)); // ================= PHOTO UPLOAD ================= const storage = multer.diskStorage({ destination: function (req, file, cb) { cb(null, "uploads/"); }, filename: function (req, file, cb) { cb(null, Date.now() + path.extname(file.originalname)); } }); const upload = multer({ storage: storage }); // ================= STUDENT SCHEMA ================= const studentSchema = new mongoose.Schema({ name: String, roll: String, branch: String, gender: String, mobile: String, email: String, photo: String }); const Student = mongoose.model("Student", studentSchema); // ================= HOME PAGE ================= app.get("/", async (req, res) => { const students = await Student.find(); let rows = ""; students.forEach(student => { rows += ` ${ student.photo ? `` : "No Photo" } ${student.name} ${student.roll} ${student.branch} ${student.gender} ${student.mobile} ${student.email}
`; }); res.send(` Student CRUD

Student Management System

+ Add Student ${rows}
Photo Name Roll Branch Gender Mobile Email Actions
`); }); // ================= ADD STUDENT FORM ================= app.get("/add", (req, res) => { res.send(`

Add Student

Name:

Roll Number:

Branch:

Gender:

Mobile:

Email:

Photo:


Back to Student List `); }); // ================= CREATE STUDENT ================= app.post("/create", upload.single("photo"), async (req, res) => { try { const student = new Student({ name: req.body.name, roll: req.body.roll, branch: req.body.branch, gender: req.body.gender, mobile: req.body.mobile, email: req.body.email, photo: req.file ? req.file.filename : "" }); await student.save(); res.redirect("/"); } catch (error) { res.status(500).send(error); } }); // ================= UPDATE FORM ================= app.get("/edit/:id", async (req, res) => { const student = await Student.findById(req.params.id); res.send(`

Update Student

${ student.photo ? `

` : "" } Name:

Roll Number:

Branch:

Gender:

Mobile:

Email:

Change Photo:


Back to Student List `); }); // ================= UPDATE STUDENT ================= app.post("/update/:id", upload.single("photo"), async (req, res) => { try { const updateData = { name: req.body.name, roll: req.body.roll, branch: req.body.branch, gender: req.body.gender, mobile: req.body.mobile, email: req.body.email }; // Update photo only if a new photo is selected if (req.file) { updateData.photo = req.file.filename; } await Student.findByIdAndUpdate( req.params.id, updateData ); res.redirect("/"); } catch (error) { res.status(500).send(error); } }); // ================= DELETE STUDENT ================= app.post("/delete/:id", async (req, res) => { try { await Student.findByIdAndDelete( req.params.id ); res.redirect("/"); } catch (error) { res.status(500).send(error); } }); // ================= START SERVER ================= app.listen(3000, () => { console.log( "Server running on http://localhost:3000" ); });
